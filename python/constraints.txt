# Pip constraints for installing sglang on newer CPython (e.g. 3.14) where
# some transitive dependencies have no prebuilt wheel yet. Applied via
# PIP_CONSTRAINT so `pip install -e python` resolves reproducibly without
# touching CI configuration.
#
# Keep this list minimal: only pin packages that fail to resolve/build on
# CPython 3.14 and that have a known-good wheel for the pinned version.
#
# NOTE: this file is intentionally dependency-free (no includes) so it can be
# consumed by any pip invocation.

# pyzmq publishes abi3 wheels; pin to a release that ships cp314 wheels.
pyzmq>=27.0.0

# pydantic v2 ships cp314 wheels from 2.11 onwards.
pydantic>=2.11.0

# regex ships cp314 wheels from 2025.9 onwards.
regex>=2025.9.1

# pybase64 ships cp314 wheels from 1.4.2 onwards.
pybase64>=1.4.2
