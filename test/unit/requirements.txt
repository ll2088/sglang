# Minimal dependency set for the run_suite unit tests.
#
# These tests only exercise run_suite.py's exit-code and collection-diagnostic
# branches; conftest.py stubs out sglang.test.ci.* so the full sglang runtime
# (torch, kernels, ...) is not needed. CI verification jobs should install this
# file instead of `pip install -e python`, which resolves the full dependency
# set and can fail during environment preparation.
#
# Usage (from the repository root):
#
#     python -m pip install -r test/unit/requirements.txt
#     python test/unit/verify_run_suite.py
#
# verify_run_suite.py is the CI entrypoint: it runs the collect-diagnostics,
# exit-codes, tests and usability dimensions and exits 2 on infra failure so
# an environment problem is never confused with a real test failure.
pytest
